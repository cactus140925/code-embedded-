function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  // Nếu là hàng đầu tiên thì thêm tiêu đề
  if (sheet.getLastRow() == 0) {
    sheet.appendRow(["Timestamp", "Temperature (°C)", "Humidity (%)", "Sunlight (%)", "LED1", "LED2"]);
  }
  
  var row = [
    new Date(),
    data.temperature,
    data.humidity,
    data.sunlight,
    data.led1 ? "ON" : "OFF",
    data.led2 ? "ON" : "OFF"
  ];
  
  sheet.appendRow(row);
  
  return ContentService.createTextOutput("OK");
}
